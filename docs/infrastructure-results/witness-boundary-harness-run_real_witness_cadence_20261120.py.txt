source=Path(sys.argv[1]).resolve();output=Path(sys.argv[2]).resolve()

assert os.name=='posix' and source==Path(manifest['path']).resolve()

def altered():return [n for n,h in manifest['files'].items() if hashlib.sha256((source/n).read_bytes()).hexdigest()!=h]
assert not altered()
started=time.monotonic();process=None
result={'schema':'owned-real-default-witness-cadence-1','source_commit':manifest['source_commit'],'completed':False,'interval_seconds':60,'max_age_seconds':180,'os_clock_changed':False,'thresholds_changed':False,'paid_api_calls':0,'real_8h_soak':False,'independent_storage_verified':False,'automatic_activation_allowed':False,'checks':[]}

def check(name,condition):
    assert condition,name
    result['checks'].append({'name':name,'passed':True,'elapsed_seconds':time.monotonic()-started})
    (output/'progress.json').write_text(json.dumps(result,indent=2)+'\n')

def guard():
    if (output/'stop.request').exists():raise InterruptedError('owned cadence stop')
    if time.monotonic()-started>600:raise TimeoutError('owned cadence deadline')

def wait_for(predicate):
    while True:
        guard()
        value=predicate()
        if value:return value
        time.sleep(.2)

def wait_until(instant):
    while datetime.now(UTC)<instant:
        guard();time.sleep(min(.2,max(0,(instant-datetime.now(UTC)).total_seconds())))

env=dict(os.environ,AWS_EC2_METADATA_DISABLED='true',AWS_SHARED_CREDENTIALS_FILE=str(output/'no-cloud-credentials'),AWS_CONFIG_FILE=str(output/'no-cloud-config'),PYTHONPYCACHEPREFIX='/home/chaehj02/.cache/zt-until0800-frozen-bytecode')
for name in ('OPENAI_API_KEY','AZURE_OPENAI_API_KEY','ANTHROPIC_API_KEY','C01_UPSTREAM_API_KEY','AWS_ACCESS_KEY_ID','AWS_SECRET_ACCESS_KEY','AWS_SESSION_TOKEN','AWS_SECURITY_TOKEN','AWS_PROFILE','AWS_DEFAULT_PROFILE','AWS_WEB_IDENTITY_TOKEN_FILE','AWS_ROLE_ARN','AWS_ROLE_SESSION_NAME','AWS_CONTAINER_CREDENTIALS_FULL_URI','AWS_CONTAINER_CREDENTIALS_RELATIVE_URI','AWS_CONTAINER_AUTHORIZATION_TOKEN'):env.pop(name,None)
env['PYTHONPATH']=str(source)
try:
    with tempfile.TemporaryDirectory(prefix='private-witness-real-cadence-') as temporary:
        root=Path(temporary);audit=SQLiteAuditStore(root/'audit.sqlite3');signer=root/'signer';control=root/'control'
        event=synthetic_events()[0];assessment=RuleBasedAnalyzer().assess(event)
        asyncio.run(audit.save(AnalysisResult(event_id=event.event_id,analyzer='rule',assessment=assessment,policy_decision=response_preview(assessment,event),latency_ms=1)))
        before=audit.path.read_bytes()
        key=witness.initialize(signer,audit.path,'real-default-cadence')['public_key_sha256']
        minimum=witness.observe(signer,audit.path,key,expected_receipt=witness.GENESIS)['receipt_sha256']
        pin=observer.initialize(control,signer,audit.path,key,expected_receipt=minimum)['profile_sha256']
        public=root/'status.json'
        profile={'profile_sha256':pin,'minimum_receipt':minimum,'status_file':str(public)}
        argv=[sys.executable,'-m','scripts.audit_witness_observer','watch','--directory',str(control),'--profile-sha256',pin,'--minimum-receipt',minimum,'--cycles','3','--status-file',str(public)]
        with (output/'watch.log').open('wb') as stream:
            process=subprocess.Popen(argv,cwd=source,env=env,stdout=stream,stderr=subprocess.STDOUT)

            wait_for(lambda: public.exists() or process.poll() is not None)
            assert public.exists()
            first=json.loads(public.read_bytes())
            check('watch_uses_real_default_policy',first['values']['interval_seconds']==60 and first['values']['max_age_seconds']==180)

        instants=[datetime.fromisoformat(r['observed_at']) for r in rows]

        (output/'expired-cache.metrics').write_text(expired)

finally:
